**Proof** (all of `test/browser/my-day*.test.ts`, 9 files)
The node clock is shifted to a local time and keeps running. Each Playwright
page starts with its Date fixed at that instant, as a real browser at that
hour would. A test that pins its own clock overrides it. The preload, saved

```js
import { createRequire } from 'node:module';
const [hh, mm] = (process.env.FAKE_LOCAL ?? '').split(':').map(Number);
const RealDate = Date;
const target = new RealDate(); target.setHours(hh, mm, 0, 0);
const offset = target.getTime() - RealDate.now();
class FakeDate extends RealDate {
  constructor(...a) { if (a.length === 0) super(RealDate.now() + offset); else super(...a); }
  static now() { return RealDate.now() + offset; }
}
globalThis.Date = FakeDate;
const { chromium } = createRequire(`${process.cwd()}/package.json`)('playwright');
const launch = chromium.launch.bind(chromium);
chromium.launch = async (...args) => {
  const browser = await launch(...args);
  const newPage = browser.newPage.bind(browser);
  browser.newPage = async (...a) => {
    const page = await newPage(...a);
    await page.clock.setFixedTime(FakeDate.now());
    return page;
  };
  return browser;
};
```
